# Scenario a (LM): V36 fresh start.
#
# AGC behaviour checked (Luminary099; pages are Luminary's):
# * PINBALL VBFRESH: V36 ENTER -> SLAP1 (FRESH_START_AND_RESTART p.211):
#   STARTSUB, FAILREG/ERCOUNT/REDOCTR = 0, lamp word keeps only NO ATT and
#   GIMBAL LOCK, DSRUPTSW = -PRIO12 (20 quick T4RUPT display passes).
# * DOFSTART (p.211-213): channel 11 = ENGINE OFF only, 12-14 zeroed, MODREG = -0 (PROG blank).
# * STARTSB2 (p.219-222): DSPOFF writes -BIT12 (blank) into DSPTAB 0..10,
#   NOUT = 11, VERBREG/NOUNREG cleared, FLAGWRD4 = 0.
# * T4RUPT QUIKDSP (p.160) sends those blanks without testing DSKYFLAG,
#   which SWINIT (p.223) leaves off -- see "Bugs found" B1.
# * V37E00E afterwards selects P00 (GOTOPOOH, p.224): PROG 00.
# Validated against yaAGC + Luminary099 (tools/oracle/bin/yaAGC).
vehicle lm
run 300                         # power-on: GOJAM -> GOPROG
expect LAMP RESTART on          # power-on GOJAM lights it (0163, yaAGC)
keys R                          # RSET: ERROR resets the RESTART lamp
expect LAMP RESTART off
keys V37E00E                    # P00
until PROG 00 within 300
keys V05N09E                    # alarm codes
until VERB 05 within 100
keys V36E                       # fresh start
run 100
expect VERB blank           # DSPOFF blanks flushed by QUIKDSP
expect PROG blank
expect FLASH off
expect LAMP KEYREL off
expect LAMP OPRERR off
expect LAMP PROG off
expect CH 11 20000              # DOFSTART: CAF BIT14, WRITE DSALMOUT
expect CH 12 00000
expect CH 14 00000
keys V                          # a key turns DSKYFLAG on: blanks go out
until R3 blank within 200       # one relay word per T4RUPT pass
expect PROG blank               # MODREG -0
expect NOUN blank
expect R1 blank
expect R2 blank
keys 05N09E                     # FAILREG cleared by SLAP1
until R3 00000 within 200
expect R1 00000
expect R2 00000
keys V37E00E                    # P00 after the fresh start
until PROG 00 within 300
expect VERB blank
expect NOUN blank
expect LAMP PROG off
