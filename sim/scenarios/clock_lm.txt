# Scenario c (LM): V06N36 shows the AGC clock; V16N36 monitors it;
# V25N36 loads it.
#
# AGC behaviour checked (Luminary099):
# * Noun 36 = ECADR TIME2, 3 components HMS, decimal only
#   (PINBALL_NOUN_TABLES p.306): R1 hours, R2 minutes, R3 seconds
#   (xxx.xx). TIME2/TIME1 count centiseconds from power-on (the sim
#   clock, which this client steps).
# * V06: HMSOUT (PINBALL p.421) reads TIME2/TIME1 once, at ENTER.
# * V16: MONITOR (p.437) re-displays through MONREQ (p.438) once per
#   second, so the display trails the clock by up to 1 s.
# * V25N36 with three ENTERs: HMSIN (p.434) converts hours, minutes,
#   seconds and stores TIME2/TIME1; the clock keeps counting from there.
# Validated against yaAGC + Luminary099 (tools/oracle/bin/yaAGC); in that
# real-time run the absolute CLOCK checks are off by the start-up latency.
vehicle lm
run 300                         # power-on
keys R V37E00E                  # RSET, P00
until PROG 00 within 300
keys V06N36E
until NOUN 36 within 100
run 50                          # R1-R3 follow, one relay word per pass
expect VERB 06
expect CLOCK ~5                 # TIME at the ENTER
keys V16N36E
run 500
expect CLOCK now ~110           # monitor: updated every second
run 1000
expect CLOCK now ~110
keys V25N36E                    # load the clock: 1 h 2 min 30.00 s
until VERB 21 within 100        # flashing: R1 load request
expect FLASH on
keys +00001E
until VERB 22 within 100
keys +00002E
until VERB 23 within 100
keys +03000E
clockref 1 2 30.00
until FLASH off within 100
keys V16N36E
run 300
expect R1 +00001
expect R2 +00002
expect CLOCK now ~110
run 6000                        # one minute later
expect R2 +00003
expect CLOCK now ~110
